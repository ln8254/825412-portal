在后端或微服务中生成凭据时，切勿使用 Math.random() 或 random.random()，它们属于伪随机数生成器 (PRNG)，其内部种子状态可被攻击者逆向还原：

Python (secrets 模块 - 密码学安全):
import secrets
import string

def generate_secure_password(length=16):
    charset = string.ascii_letters + string.digits + "!@#$%^&*()_+-="
    return ''.join(secrets.choice(charset) for _ in range(length))

# 输出示例: K9#xP2$vL9@mQ8*z
print(generate_secure_password(16))
Node.js (crypto.randomBytes / Web Crypto API):
const crypto = require('crypto');

function generateSecurePassword(length = 16) {
  const charset = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#$%^&*()_+-=';
  const randomBytes = crypto.randomBytes(length);
  let result = '';
  for (let i = 0; i < length; i++) {
    result += charset[randomBytes[i] % charset.length];
  }
  return result;
}

console.log(generateSecurePassword(16));
